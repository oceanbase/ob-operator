Database SQL access
{{- $host := "" }}
{{- if eq .Values.sqlService.type "NodePort" }}
{{- $host = include "oceanseek.accessHost" (dict "host" .Values.access.host "namespace" .Release.Namespace "service" (printf "%s-sql" .Release.Name)) }}
{{- end }}
{{- if $host }}
{{- $port := int (include "oceanseek.sqlNodePort" .) }}
{{- if $port }}
  mysql -h '{{ replace "'" "'\"'\"'" $host }}' -P {{ $port }} -u 'root@sys' --protocol=TCP -p
{{- else }}
  SQL_PORT="$(kubectl -n {{ .Release.Namespace }} get svc {{ .Release.Name }}-sql -o jsonpath='{.spec.ports[0].nodePort}')" && test -n "$SQL_PORT" && mysql -h '{{ replace "'" "'\"'\"'" $host }}' -P "$SQL_PORT" -u 'root@sys' --protocol=TCP -p
{{- end }}
{{- else }}
Inside Kubernetes:
  mysql -h {{ .Release.Name }}-sql.{{ .Release.Namespace }}.svc -P {{ .Values.sqlService.port }} -u 'root@sys' -p
For local access, keep the forward running and connect in a second terminal:
  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ .Release.Name }}-sql 2883:{{ .Values.sqlService.port }}
  mysql -h 127.0.0.1 -P 2883 -u 'root@sys' -p
{{- end }}
Root password (in a trusted terminal):
  kubectl -n {{ .Release.Namespace }} get secret {{ .Values.observer.rootExistingSecret | default (printf "%s-db-root" .Release.Name) }} -o jsonpath='{.data.password}' | base64 --decode; echo
{{- else -}}
SQL access becomes available after stage=cluster.
{{- end }}
